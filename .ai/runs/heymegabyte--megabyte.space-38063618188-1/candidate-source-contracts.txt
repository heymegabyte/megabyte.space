packages:
  - packages/*
  # The shared build tooling as `@gadgets/scripts`, so packages reach it by name rather than by a
  # relative path that breaks in forks vendoring this repo. Top level: it resolves from the root.
  - scripts

# The toolchain versions every package shares. Declared once here so a bump is one edit rather than
# needing to hand edit a bunch of different package.json files.
catalog:
  '@cloudflare/vitest-pool-workers': ^0.22.0
  '@cloudflare/workers-types': ^5.20261001.1
  capnweb: ^0.12.0
  # jsdiff: line diffs in workshop-backend (formatUnifiedDiff) and gatekeeper-kit's git-diff (the
  # simulated PR/MR diff's hunks, used by the git-host gatekeepers). One catalog entry so the two
  # can't drift apart.
  diff: ^8.0.4
  # Exact: pinned in lockstep with capnweb (declared as a >=0.7.0 peer)
  capnweb-validate: 0.3.0
  # Exact: a prerelease, kept in step with the Wrangler/workerd stack the test pool runs on.
  miniflare: 5.20261001.0-alpha
  # Exact: TypeScript 7 (tsgo). Type-checking runs on the native compiler. Build-time
  # transpilers that need the JS compiler API (scripts/build-gatekeeper-configurator.ts)
  # use the "typescript6" npm alias below instead; capnweb-validate
  # ships its own JS-based compiler dependency since 0.2.4.
  typescript: 7.0.2
  # Exact: the JS compiler API ("typescript" 7 no longer ships it), aliased so it can sit beside tsgo.
  typescript6: npm:typescript@6.0.3
  # Exact, not ranged: the test transform has to stay on esbuild. Vite 8's Oxc does not lower the
  # Stage-3 decorators capnweb-validate's `@validateRpc()` rewrites to (oxc-project/oxc#9170), so the
  # raw `@` reaches workerd and every suite importing a decorated class fails to load with
  # "SyntaxError: Invalid or unexpected token" (cloudflare/workers-sdk#12626; re-verified on 8.3.1).
  # Oxc's `decorator: { legacy: true }` does load them, but with TypeScript-legacy semantics that
  # differ from the Stage-3 lowering wrangler deploys, so it only works while capnweb-validate's
  # decorators ignore their `context` argument.
  vite: 7.3.6
  vite-plus: ^1.0.0
  vitest: ^4.1.11
  wrangler: ^4.147.0
  zod: ^4.6.5

overrides:
  # Keep the test pool's Wrangler/Miniflare/workerd stack aligned with the workspace toolchain.
  '@cloudflare/vitest-pool-workers>miniflare': "catalog:"
  '@cloudflare/vitest-pool-workers>wrangler': "catalog:"
  # Vitest peers can resolve vite@latest, so the version is pinned rather than ranged. Points at the
  # catalog so this and the manifests can't disagree: a catalog only covers what a workspace
  # manifest declares itself, and the override is what reaches vite as somebody else's peer.
  # Selected by range, not bare `vite`: vite-plus 1.0 depends on `vite` as an alias for
  # `npm:@voidzero-dev/vite-plus-core`, which a semver selector does not match, so `vp` keeps the
  # core it was built against instead of having vite 7 substituted under it.
  vite@*: "catalog:"
  # minimumReleaseAge gates direct deps but not these two, so pin them to the last
  # mature release (CI verifies every lockfile entry against the release-age policy):
  # - @types/node arrives as an optional vite/vitest peer resolved to `latest`.
  # - @lezer/markdown is a deep transitive of @codemirror/lang-markdown.
  '@types/node': 26.1.0
  '@lezer/markdown': 1.6.4

# Reject dependency versions published within this many minutes (24h), matching the
# CI supply-chain policy so local installs can't commit a too-fresh lockfile.
